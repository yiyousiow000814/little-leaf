# Staged diagnostic only. Never included in a release project.
var depth_probe_active=false
var depth_probe_events=[]
var depth_probe_context={}
var depth_probe_dropped=0
func _depth_probe_reset():
	depth_probe_events=[];depth_probe_context={};depth_probe_dropped=0
func _depth_probe_begin(part:String,id:String,world:Vector2,depth:float):
	depth_probe_context={}
	if not depth_probe_active:return
	if id!="qa_depth_actor" and id not in ["101","102"] and not (part=="scenery_tree" and world.distance_to(Vector2(-13.1,1.5))<.01):return
	var anchor=iso(world.x,world.y)+_presentation_offset
	depth_probe_context={"id":id,"part":part,"world":[world.x,world.y],"anchor":[anchor.x,anchor.y],"depth":depth,"cursor_begin":canvas_stream.cursor}
	depth_probe_events.append({"kind":"begin","context":depth_probe_context.duplicate(true)})
func _depth_probe_entity(e:Dictionary):
	var world:Vector2=e.get("position",Vector2(float(e.entry.get("x",0))+.5,float(e.entry.get("z",0))+.5))
	world+=e.get("meal_offset",Vector2.ZERO)
	_depth_probe_begin(str(e.type),str(e.entry.get("id","")),world,float(e.depth))
func _depth_probe_submit(kind:String,order:int,node:Node2D):
	if not depth_probe_active or depth_probe_context.is_empty():return
	if depth_probe_events.size()>=128:depth_probe_dropped+=1;return
	depth_probe_events.append({"kind":kind,"context":depth_probe_context.duplicate(true),"cursor":order,"node_id":node.get_instance_id(),"index_at_submit":node.get_index(),"parent_id":node.get_parent().get_instance_id(),"generation":canvas_stream.frame_id})
func _depth_probe_finish():
	if not depth_probe_active:return
	for row in depth_probe_events:
		if not row.has("node_id"):continue
		var node=instance_from_id(row.node_id)
		row["final_index"]=node.get_index() if is_instance_valid(node) else -1
		row["visible"]=node.visible if is_instance_valid(node) else false
